// A 2-layer neural network in pure C
float sigmoid(float x) {
    return 1.0f / (1.0f + expf(-x));
}

void forward_pass(Network *net, float *input) {
    for (int i = 0; i < net->hidden_size; i++) {
        float sum = net->bias1[i];
        for (int j = 0; j < net->input_size; j++)
            sum += input[j] * net->w1[j][i];
        net->hidden[i] = sigmoid(sum);
    }
    float out = net->bias2;
    for (int i = 0; i < net->hidden_size; i++)
        out += net->hidden[i] * net->w2[i];
    net->output = sigmoid(out);
}
