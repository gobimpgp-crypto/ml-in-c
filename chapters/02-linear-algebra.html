Why Linear Algebra?

Almost every ML operation boils down to:

  • Vectors — a data point, a weight vector, a gradient
  • Matrix multiplication — a forward pass through a neural network layer
  • Dot products — measuring similarity, computing scores

We'll build a minimal linear algebra library in C that we'll reuse in every subsequent chapter.

Vectors

Representation

A vector is just a 1D array. We'll use a struct to keep the size with the data:

#include <stdio.h>
#include <stdlib.h>
#include <math.h>

typedef struct {
    float *data;
    int    size;
} Vec;

Vec *vec_create(int size) {
    Vec *v = malloc(sizeof(Vec));
    v->data = calloc(size, sizeof(float));  // zero-initialized
    v->size = size;
    return v;
}

void vec_free(Vec *v) {
    free(v->data);
    free(v);
}

void vec_print(const Vec *v) {
    printf("[");
    for (int i = 0; i < v->size; i++)
        printf(i == v->size-1 ? "%.4f" : "%.4f, ", v->data[i]);
    printf("]\n");
}

Vector Operations

// Element-wise addition: c = a + b
Vec *vec_add(const Vec *a, const Vec *b) {
    Vec *c = vec_create(a->size);
    for (int i = 0; i < a->size; i++)
        c->data[i] = a->data[i] + b->data[i];
    return c;
}

// Scalar multiply: b = alpha * a
Vec *vec_scale(const Vec *a, float alpha) {
    Vec *b = vec_create(a->size);
    for (int i = 0; i < a->size; i++)
        b->data[i] = a->data[i] * alpha;
    return b;
}

// Dot product: scalar = a · b
float vec_dot(const Vec *a, const Vec *b) {
    float sum = 0.0f;
    for (int i = 0; i < a->size; i++)
        sum += a->data[i] * b->data[i];
    return sum;
}

// L2 norm: ||v|| = sqrt(sum of squares)
float vec_norm(const Vec *v) {
    return sqrtf(vec_dot(v, v));
}

The Dot Product — Geometric Meaning

The dot product a · b = ||a|| ||b|| cos(θ) measures how "aligned" two vectors are:

  • a · b > 0 — vectors point in a similar direction
  • a · b = 0 — vectors are perpendicular (orthogonal)
  • a · b < 0 — vectors point in opposite directions
a · b = a₀b₀ + a₁b₁ + ... + aₙbₙ
int main(void) {
    Vec *a = vec_create(3);
    Vec *b = vec_create(3);

    float a_vals[] = {1.0f, 2.0f, 3.0f};
    float b_vals[] = {4.0f, 5.0f, 6.0f};
    for (int i = 0; i < 3; i++) {
        a->data[i] = a_vals[i];
        b->data[i] = b_vals[i];
    }

    printf("a = "); vec_print(a);
    printf("b = "); vec_print(b);
    printf("a · b = %.2f\n", vec_dot(a, b));   // 32.00
    printf("||a|| = %.4f\n", vec_norm(a));       // 3.7417

    vec_free(a); vec_free(b);
    return 0;
}

Matrices

Representation

We store matrices in row-major order — one contiguous array with row * cols + col indexing:

typedef struct {
    float *data;
    int    rows;
    int    cols;
} Mat;

Mat *mat_create(int rows, int cols) {
    Mat *m = malloc(sizeof(Mat));
    m->data = calloc(rows * cols, sizeof(float));
    m->rows = rows;
    m->cols = cols;
    return m;
}

// Inline access macro
#define MAT(m, r, c) ((m)->data[(r) * (m)->cols + (c)])

void mat_free(Mat *m) {
    free(m->data);
    free(m);
}

void mat_print(const Mat *m) {
    for (int r = 0; r < m->rows; r++) {
        printf("[ ");
        for (int c = 0; c < m->cols; c++)
            printf("%8.4f ", MAT(m, r, c));
        printf("]\n");
    }
}

Matrix Multiplication

The heart of every neural network forward pass. For matrices A (m×k) and B (k×n), result C (m×n):

C[i][j] = Σₖ A[i][k] × B[k][j]
// C = A × B  (A is m×k, B is k×n, C is m×n)
Mat *mat_mul(const Mat *A, const Mat *B) {
    // A->cols must equal B->rows
    Mat *C = mat_create(A->rows, B->cols);

    for (int i = 0; i < A->rows; i++) {
        for (int j = 0; j < B->cols; j++) {
            float sum = 0.0f;
            for (int k = 0; k < A->cols; k++)
                sum += MAT(A, i, k) * MAT(B, k, j);
            MAT(C, i, j) = sum;
        }
    }
    return C;
}
⚠ O(n³) Complexity Naive matrix multiplication is O(n³). For large matrices, real engines use BLAS routines (dgemm/sgemm). For learning, naive is perfect.

Matrix Transpose

Transposing swaps rows and columns. Used constantly in backpropagation:

// Aᵀ[j][i] = A[i][j]
Mat *mat_transpose(const Mat *A) {
    Mat *T = mat_create(A->cols, A->rows);
    for (int i = 0; i < A->rows; i++)
        for (int j = 0; j < A->cols; j++)
            MAT(T, j, i) = MAT(A, i, j);
    return T;
}

// Matrix-vector multiply: y = A × x
Vec *mat_vec_mul(const Mat *A, const Vec *x) {
    Vec *y = vec_create(A->rows);
    for (int i = 0; i < A->rows; i++) {
        float sum = 0.0f;
        for (int j = 0; j < A->cols; j++)
            sum += MAT(A, i, j) * x->data[j];
        y->data[i] = sum;
    }
    return y;
}

Putting It Together: A Complete Example

#include <stdio.h>
#include <stdlib.h>
#include <math.h>

// ... (include all the structs and functions above)

int main(void) {
    // Create a 2×3 matrix A
    Mat *A = mat_create(2, 3);
    MAT(A,0,0)=1; MAT(A,0,1)=2; MAT(A,0,2)=3;
    MAT(A,1,0)=4; MAT(A,1,1)=5; MAT(A,1,2)=6;

    // Create a 3×2 matrix B
    Mat *B = mat_create(3, 2);
    MAT(B,0,0)=7;  MAT(B,0,1)=8;
    MAT(B,1,0)=9;  MAT(B,1,1)=10;
    MAT(B,2,0)=11; MAT(B,2,1)=12;

    printf("A =\n"); mat_print(A);
    printf("B =\n"); mat_print(B);

    Mat *C = mat_mul(A, B);
    printf("A × B =\n"); mat_print(C);
    // [ 58.00  64.00 ]
    // [139.00 154.00 ]

    mat_free(A); mat_free(B); mat_free(C);
    return 0;
}

Element-wise Operations

ML also needs element-wise ops — applying a function to every element:

// Apply a function to every element in-place
void mat_apply(Mat *m, float (*fn)(float)) {
    int total = m->rows * m->cols;
    for (int i = 0; i < total; i++)
        m->data[i] = fn(m->data[i]);
}

// Example: apply ReLU element-wise
float relu(float x) { return x > 0.0f ? x : 0.0f; }

// In practice:
// mat_apply(layer_output, relu);

Summary

  • Vectors = 1D float arrays; matrices = 2D stored in row-major 1D arrays
  • Dot product: Σ aᵢbᵢ — measures alignment
  • Matrix multiply: O(n³) triple loop, result shape is (A.rows × B.cols)
  • Transpose swaps rows/cols — essential for gradient computation
  • Use #define MAT(m,r,c) macros for clean 2D indexing
