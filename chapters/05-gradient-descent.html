The Core Idea

Imagine you're blindfolded on a hilly landscape and want to reach the valley. At each step, you feel the slope under your feet and take a step downhill. That's gradient descent:

  1. Compute the gradient of the loss with respect to each parameter
  2. Step each parameter in the opposite direction of its gradient
  3. Repeat until the loss stops decreasing
w ← w - α × ∂J/∂w b ← b - α × ∂J/∂b

where α is the learning rate — how big each step is.

Gradients for Linear Regression

For MSE loss J = (1/2n) Σ (ŷᵢ - yᵢ)², the gradients are:

∂J/∂wⱼ = (1/n) × Σᵢ (ŷᵢ - yᵢ) × xᵢⱼ ∂J/∂b = (1/n) × Σᵢ (ŷᵢ - yᵢ)

Batch Gradient Descent

Uses all training samples to compute one gradient update:

#include <stdio.h>
#include <stdlib.h>
#include <math.h>
#include <string.h>

typedef struct {
    float *w;    // weights [n_features]
    float  b;    // bias
    int    p;    // number of features
} Model;

// Compute prediction for one sample
float predict(const Model *m, const float *x) {
    float s = m->b;
    for (int j = 0; j < m->p; j++) s += m->w[j] * x[j];
    return s;
}

// Compute MSE
float mse(const Model *m, const float *X, const float *y, int n) {
    float s = 0;
    for (int i = 0; i < n; i++) {
        float d = predict(m, X + i*m->p) - y[i];
        s += d*d;
    }
    return s / n;
}

// One step of batch gradient descent
// X: [n × p], y: [n], lr: learning rate
void gradient_step(Model *m, const float *X, const float *y,
                   int n, float lr) {
    float *dw = calloc(m->p, sizeof(float));
    float  db = 0.0f;

    for (int i = 0; i < n; i++) {
        const float *xi = X + i * m->p;
        float error = predict(m, xi) - y[i];   // ŷ - y
        for (int j = 0; j < m->p; j++)
            dw[j] += error * xi[j];
        db += error;
    }
    // Average and update
    for (int j = 0; j < m->p; j++)
        m->w[j] -= lr * dw[j] / n;
    m->b -= lr * db / n;

    free(dw);
}

// Training loop
void train(Model *m, const float *X, const float *y,
           int n, float lr, int epochs) {
    printf("Epoch   MSE\n");
    for (int e = 0; e < epochs; e++) {
        gradient_step(m, X, y, n, lr);
        if (e % 100 == 0 || e == epochs-1) {
            printf("%5d   %.6f\n", e, mse(m, X, y, n));
        }
    }
}

int main(void) {
    // y = 3x₁ + 2x₂ + 1 + noise
    float X[] = {
        1.0f, 0.5f,
        2.0f, 1.0f,
        3.0f, 1.5f,
        4.0f, 2.0f,
        5.0f, 2.5f,
    };
    float y[] = {5.0f, 9.2f, 13.8f, 18.1f, 22.9f};
    int n = 5, p = 2;

    Model m = { .w = calloc(p, sizeof(float)), .b = 0.0f, .p = p };

    train(&m, X, y, n, 0.01f, 1000);
    printf("\nLearned weights: w=[%.3f, %.3f] b=%.3f\n",
           m.w[0], m.w[1], m.b);
    // Should be close to w=[3, 2] b=1

    free(m.w);
    return 0;
}

Stochastic Gradient Descent (SGD)

Instead of all n samples per step, use just one random sample. Noisy but fast:

#include <stdlib.h>

// One SGD step using a single random sample
void sgd_step(Model *m, const float *X, const float *y,
              int n, float lr) {
    int i = rand() % n;
    const float *xi = X + i * m->p;
    float error = predict(m, xi) - y[i];

    for (int j = 0; j < m->p; j++)
        m->w[j] -= lr * error * xi[j];
    m->b -= lr * error;
}

// SGD training loop
void train_sgd(Model *m, const float *X, const float *y,
               int n, float lr, int steps) {
    srand(42);
    for (int s = 0; s < steps; s++) {
        sgd_step(m, X, y, n, lr);
        if (s % 500 == 0)
            printf("Step %d: MSE = %.6f\n", s, mse(m, X, y, n));
    }
}

Mini-Batch Gradient Descent

Best of both worlds: use a random batch of size B on each step. Standard in modern deep learning:

// Fisher-Yates shuffle for random batches
void shuffle(int *idx, int n) {
    for (int i = n-1; i > 0; i--) {
        int j = rand() % (i+1);
        int tmp = idx[i]; idx[i] = idx[j]; idx[j] = tmp;
    }
}

void minibatch_step(Model *m, const float *X, const float *y,
                    int n, float lr, int batch_size,
                    const int *batch_idx) {
    float *dw = calloc(m->p, sizeof(float));
    float  db = 0.0f;

    for (int k = 0; k < batch_size; k++) {
        int i = batch_idx[k];
        const float *xi = X + i * m->p;
        float error = predict(m, xi) - y[i];
        for (int j = 0; j < m->p; j++)
            dw[j] += error * xi[j];
        db += error;
    }
    for (int j = 0; j < m->p; j++)
        m->w[j] -= lr * dw[j] / batch_size;
    m->b -= lr * db / batch_size;
    free(dw);
}

void train_minibatch(Model *m, const float *X, const float *y,
                     int n, float lr, int epochs, int batch_size) {
    int *idx = malloc(n * sizeof(int));
    for (int i = 0; i < n; i++) idx[i] = i;

    for (int e = 0; e < epochs; e++) {
        shuffle(idx, n);
        for (int start = 0; start < n; start += batch_size) {
            int bsz = (start + batch_size < n) ? batch_size : n - start;
            minibatch_step(m, X, y, n, lr, bsz, idx + start);
        }
        if (e % 50 == 0)
            printf("Epoch %d: MSE = %.6f\n", e, mse(m, X, y, n));
    }
    free(idx);
}

Learning Rate — The Critical Hyperparameter

⚠ Too Large or Too Small? If lr is too large, gradient descent diverges (loss goes up). If too small, training is painfully slow. A good starting point: try 0.01, 0.001, and 0.0001. 
// Learning rate schedule: decay by factor each epoch
float lr_schedule(float lr0, float decay, int epoch) {
    return lr0 / (1.0f + decay * epoch);
}

Convergence Check

// Stop when the improvement is below tolerance
void train_until_convergence(Model *m, const float *X, const float *y,
                             int n, float lr, float tol, int max_epochs) {
    float prev_loss = 1e30f;
    for (int e = 0; e < max_epochs; e++) {
        gradient_step(m, X, y, n, lr);
        float loss = mse(m, X, y, n);
        if (e % 100 == 0)
            printf("Epoch %d: loss=%.6f\n", e, loss);
        if (fabsf(prev_loss - loss) < tol) {
            printf("Converged at epoch %d\n", e);
            break;
        }
        prev_loss = loss;
    }
}

Comparison: Batch vs SGD vs Mini-Batch

Which to use?
  • Batch GD — Smooth convergence, exact gradient. Slow for large datasets.
  • SGD — Very fast per step. Noisy — can escape local minima. Works for online learning.
  • Mini-Batch — Best of both. Use batch_size=32 or 64. Standard for deep learning.

Summary

  • Gradient descent: w ← w - α × ∂J/∂w repeated until convergence
  • Batch GD: exact gradient, O(n) per step
  • SGD: one random sample per step — noisy but fast
  • Mini-batch: random batch of size B — the deep learning standard
  • Learning rate is critical — too high diverges, too low is slow
