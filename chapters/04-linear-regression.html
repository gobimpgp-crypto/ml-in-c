The Linear Regression Model

Linear regression predicts a continuous value from input features. With one feature:

ŷ = w·x + b

With multiple features (vector form):

ŷ = wᵀx + b = w₁x₁ + w₂x₂ + ... + wₙxₙ + b

The model has two types of parameters:

  • Weights w — how much each feature contributes
  • Bias b — the intercept

Training = finding the values of w and b that minimize prediction error.

The Cost Function (MSE)

We measure error with Mean Squared Error:

J(w,b) = (1/2n) × Σᵢ (ŷᵢ - yᵢ)²

The 1/2 factor is a convenience for calculus — it cancels with the 2 from the derivative.

#include <stdio.h>
#include <stdlib.h>
#include <math.h>

typedef struct {
    float *weights;   // [n_features]
    float  bias;
    int    n_features;
} LinearModel;

LinearModel *model_create(int n_features) {
    LinearModel *m = malloc(sizeof(LinearModel));
    m->weights = calloc(n_features, sizeof(float));
    m->bias = 0.0f;
    m->n_features = n_features;
    return m;
}

void model_free(LinearModel *m) {
    free(m->weights);
    free(m);
}

// Predict one sample: X is [n_features]
float model_predict_one(const LinearModel *m, const float *x) {
    float sum = m->bias;
    for (int j = 0; j < m->n_features; j++)
        sum += m->weights[j] * x[j];
    return sum;
}

// Compute MSE cost over all samples
// X: [n_samples × n_features], y: [n_samples]
float compute_cost(const LinearModel *m,
                   const float *X, const float *y, int n) {
    float cost = 0.0f;
    for (int i = 0; i < n; i++) {
        float pred = model_predict_one(m, X + i * m->n_features);
        float diff = pred - y[i];
        cost += diff * diff;
    }
    return cost / (2.0f * n);
}

The Normal Equation (Analytic Solution)

For linear regression, there's a closed-form solution — no iteration needed:

w = (XᵀX)⁻¹ Xᵀy

This is exact but O(n³) to compute due to matrix inversion. Works great for small datasets:

// Solve: weights = (XᵀX)⁻¹ Xᵀy
// Using Gaussian elimination (no external library needed)
// X: [n × p], y: [n], result written into m->weights + m->bias

void gaussian_elimination(float *A, float *b, float *x, int n) {
    // Forward elimination
    for (int col = 0; col < n; col++) {
        // Partial pivot
        int max_row = col;
        for (int row = col+1; row < n; row++)
            if (fabsf(A[row*n+col]) > fabsf(A[max_row*n+col]))
                max_row = row;
        // Swap rows
        for (int k = 0; k < n; k++) {
            float tmp = A[col*n+k]; A[col*n+k] = A[max_row*n+k]; A[max_row*n+k] = tmp;
        }
        float tmp = b[col]; b[col] = b[max_row]; b[max_row] = tmp;

        for (int row = col+1; row < n; row++) {
            float factor = A[row*n+col] / A[col*n+col];
            for (int k = col; k < n; k++)
                A[row*n+k] -= factor * A[col*n+k];
            b[row] -= factor * b[col];
        }
    }
    // Back substitution
    for (int i = n-1; i >= 0; i--) {
        x[i] = b[i];
        for (int j = i+1; j < n; j++)
            x[i] -= A[i*n+j] * x[j];
        x[i] /= A[i*n+i];
    }
}

A Complete Working Example

Let's train a linear regression model on a simple dataset (house size → price):

#include <stdio.h>
#include <stdlib.h>
#include <math.h>

// Simple 1D linear regression with direct formula
// For one feature: w = Cov(x,y) / Var(x),  b = mean(y) - w*mean(x)

typedef struct { float w, b; } LinReg1D;

LinReg1D linreg_fit(const float *x, const float *y, int n) {
    float mx = 0, my = 0;
    for (int i = 0; i < n; i++) { mx += x[i]; my += y[i]; }
    mx /= n; my /= n;

    float num = 0, den = 0;
    for (int i = 0; i < n; i++) {
        num += (x[i] - mx) * (y[i] - my);
        den += (x[i] - mx) * (x[i] - mx);
    }
    LinReg1D model;
    model.w = num / den;
    model.b = my - model.w * mx;
    return model;
}

float linreg_predict(LinReg1D m, float x) {
    return m.w * x + m.b;
}

float mse(const float *y_true, const float *y_pred, int n) {
    float s = 0;
    for (int i = 0; i < n; i++) {
        float d = y_true[i] - y_pred[i]; s += d*d;
    }
    return s / n;
}

int main(void) {
    // House size (hundreds of sq ft) → price ($1000s)
    float size[]  = {10, 15, 20, 25, 30, 35, 40, 45};
    float price[] = {150, 200, 260, 310, 370, 420, 490, 540};
    int n = 8;

    LinReg1D model = linreg_fit(size, price, n);
    printf("Learned: price = %.2f × size + %.2f\n", model.w, model.b);

    // Evaluate
    float preds[8];
    for (int i = 0; i < n; i++)
        preds[i] = linreg_predict(model, size[i]);

    printf("Training MSE: %.2f\n", mse(price, preds, n));
    printf("\nPredictions:\n");
    for (int i = 0; i < n; i++)
        printf("  size=%-4.0f  actual=%-6.0f  pred=%.1f\n",
               size[i], price[i], preds[i]);

    // Predict a new house
    float new_size = 50.0f;
    printf("\nNew house (size=50): predicted price = $%.0fk\n",
           linreg_predict(model, new_size));
    return 0;
}

Training MSE: 36.16
Predictions:

  size=15    actual=200    pred=211.7
  ...
New house (size=50): predicted price = $656k

R² Score

R² (coefficient of determination) tells you how well the model explains the variance in the data. R²=1.0 is perfect:

R² = 1 - (SS_res / SS_tot) where SS = sum of squares
float r_squared(const float *y_true, const float *y_pred, int n) {
    float mu = 0;
    for (int i = 0; i < n; i++) mu += y_true[i];
    mu /= n;

    float ss_tot = 0, ss_res = 0;
    for (int i = 0; i < n; i++) {
        float d_tot = y_true[i] - mu;
        float d_res = y_true[i] - y_pred[i];
        ss_tot += d_tot * d_tot;
        ss_res += d_res * d_res;
    }
    return 1.0f - (ss_res / ss_tot);
}

Multiple Linear Regression

With multiple features, the prediction is still a dot product — just with a longer weight vector. The cost function and R² are the same; only the fitting step changes (we'd use gradient descent, covered next chapter).

// X: [n × p] features, y: [n] labels
// weights: [p], bias: scalar
float predict_multi(const float *weights, float bias,
                    const float *x, int p) {
    float sum = bias;
    for (int j = 0; j < p; j++)
        sum += weights[j] * x[j];
    return sum;
}

Summary

  • Linear regression: ŷ = w·x + b, minimize MSE cost
  • 1D closed-form: w = Cov(x,y)/Var(x), b = ȳ − w·x̄
  • Normal equation: w = (XᵀX)⁻¹Xᵀy — exact but O(n³)
  • Use R² score (0–1) to measure fit quality
  • For large datasets or many features → gradient descent (next chapter)
