From Linear to Non-Linear

Linear regression can only draw a straight line. Real-world data (images, audio, language) has non-linear patterns. Neural networks stack linear transformations + non-linear activations to approximate any function.

A single neuron computes:

output = activation(w · x + b)

A layer is just many neurons in parallel. Stack layers → deep network.

Activation Functions

Activation functions introduce non-linearity. Without them, stacking layers is equivalent to just one linear layer.

#include <math.h>

// Sigmoid: maps any value to (0, 1) — good for output probabilities
float sigmoid(float x) {
    return 1.0f / (1.0f + expf(-x));
}
float sigmoid_deriv(float x) {
    float s = sigmoid(x);
    return s * (1.0f - s);
}

// ReLU: max(0, x) — most common in hidden layers
float relu(float x) {
    return x > 0.0f ? x : 0.0f;
}
float relu_deriv(float x) {
    return x > 0.0f ? 1.0f : 0.0f;
}

// Leaky ReLU: fixes "dying ReLU" problem
float leaky_relu(float x, float alpha) {
    return x > 0.0f ? x : alpha * x;
}
float leaky_relu_deriv(float x, float alpha) {
    return x > 0.0f ? 1.0f : alpha;
}

// Tanh: maps to (-1, 1) — centered, good for recurrent layers
float tanh_act(float x) {
    return tanhf(x);
}
float tanh_deriv(float x) {
    float t = tanhf(x);
    return 1.0f - t * t;
}

// Softmax: for multi-class output (converts logits to probabilities)
void softmax(float *x, int n) {
    float max_val = x[0];
    for (int i = 1; i < n; i++) if (x[i] > max_val) max_val = x[i];
    float sum = 0.0f;
    for (int i = 0; i < n; i++) { x[i] = expf(x[i] - max_val); sum += x[i]; }
    for (int i = 0; i < n; i++) x[i] /= sum;
}

Layer Structure

#include <stdlib.h>
#include <string.h>

typedef enum { RELU, SIGMOID, TANH, LINEAR } ActivationType;

typedef struct {
    float *W;        // weights [out_size × in_size]
    float *b;        // biases  [out_size]
    float *z;        // pre-activation  [out_size]
    float *a;        // post-activation [out_size]
    float *dW;       // weight gradients
    float *db;       // bias gradients
    int    in_size;
    int    out_size;
    ActivationType act;
} Layer;

Layer *layer_create(int in_size, int out_size, ActivationType act) {
    Layer *l = malloc(sizeof(Layer));
    l->in_size  = in_size;
    l->out_size = out_size;
    l->act      = act;

    l->W  = malloc(out_size * in_size * sizeof(float));
    l->b  = calloc(out_size, sizeof(float));
    l->z  = malloc(out_size * sizeof(float));
    l->a  = malloc(out_size * sizeof(float));
    l->dW = calloc(out_size * in_size, sizeof(float));
    l->db = calloc(out_size, sizeof(float));

    // He initialization for ReLU
    float std = sqrtf(2.0f / in_size);
    srand(42);
    for (int i = 0; i < out_size * in_size; i++)
        l->W[i] = ((float)rand()/RAND_MAX * 2 - 1) * std;

    return l;
}

void layer_free(Layer *l) {
    free(l->W); free(l->b); free(l->z); free(l->a);
    free(l->dW); free(l->db); free(l);
}

// Apply activation in-place on l->z → l->a
void apply_activation(Layer *l) {
    for (int i = 0; i < l->out_size; i++) {
        switch (l->act) {
            case RELU:    l->a[i] = relu(l->z[i]);    break;
            case SIGMOID: l->a[i] = sigmoid(l->z[i]); break;
            case TANH:    l->a[i] = tanh_act(l->z[i]);break;
            case LINEAR:  l->a[i] = l->z[i];          break;
        }
    }
}

Forward Pass

For each layer, compute z = W·a_prev + b, then a = activation(z):

// Forward pass through one layer
// input: [in_size], output written to l->a
void layer_forward(Layer *l, const float *input) {
    for (int i = 0; i < l->out_size; i++) {
        float sum = l->b[i];
        for (int j = 0; j < l->in_size; j++)
            sum += l->W[i * l->in_size + j] * input[j];
        l->z[i] = sum;
    }
    apply_activation(l);
}

Full Neural Network

typedef struct {
    Layer **layers;
    int     n_layers;
} Network;

Network *net_create(int *sizes, ActivationType *acts, int n_layers) {
    Network *net = malloc(sizeof(Network));
    net->n_layers = n_layers;
    net->layers = malloc(n_layers * sizeof(Layer *));
    for (int i = 0; i < n_layers; i++)
        net->layers[i] = layer_create(sizes[i], sizes[i+1], acts[i]);
    return net;
}

void net_free(Network *net) {
    for (int i = 0; i < net->n_layers; i++) layer_free(net->layers[i]);
    free(net->layers); free(net);
}

// Forward pass through the entire network
// input: [input_size], returns pointer to output layer activations
float *net_forward(Network *net, const float *input) {
    const float *prev = input;
    for (int i = 0; i < net->n_layers; i++) {
        layer_forward(net->layers[i], prev);
        prev = net->layers[i]->a;
    }
    return net->layers[net->n_layers - 1]->a;
}

// Example: 2-layer network (2 inputs, 4 hidden, 1 output)
int main(void) {
    int sizes[]          = {2, 4, 1};
    ActivationType acts[] = {RELU, SIGMOID};
    Network *net = net_create(sizes, acts, 2);

    float input[] = {0.5f, 0.8f};
    float *output = net_forward(net, input);
    printf("Output: %.4f\n", output[0]);

    net_free(net);
    return 0;
}
✓ What just happened? We fed [0.5, 0.8] into the network. It computed activations through a hidden ReLU layer and a sigmoid output layer — a full forward pass, in ~100 lines of C.

Universal Approximation Theorem

A neural network with even a single hidden layer and enough neurons can approximate any continuous function. More layers = more efficient representation of complex functions.

Choosing Architecture

  • Input size = number of features
  • Output size = 1 for regression, number of classes for classification
  • Hidden layers: 1–3 layers of 32–256 units works for most tasks
  • Output activation: linear for regression, sigmoid for binary classification, softmax for multi-class

Summary

  • A neuron = linear combination + activation function
  • ReLU is the go-to for hidden layers; sigmoid/softmax for outputs
  • Forward pass: flow input through each layer's z = Wx + b, then activation
  • Use He init for ReLU layers (std = √(2/fan_in))
  • The Layer struct stores weights, biases, and pre-allocated activation buffers
